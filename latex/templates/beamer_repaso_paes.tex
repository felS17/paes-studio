  \paso{1}{$4^3 = (2^2)^3 = 2^6$ y $8^3 = (2^3)^3 = 2^9$.}
  \paso{2}{$2^5\cdot 2^6 = 2^{11}$.}
  \paso{3}{$\dfrac{2^{11}}{2^{9}} = 2^2 = 4$.}
  \vspace{0.4em}
