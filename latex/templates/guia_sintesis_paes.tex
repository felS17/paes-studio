\section{Cómo usar esta guía}
Cada sección corresponde a un eje temático, ordenado según su frecuencia en los ensayos.
Lee primero las cajas de definición y fórmula; luego resuelve el ejemplo antes de mirar el desarrollo.

\begin{definicion}[Potencias de igual base]
Para $a \neq 0$ y $m, n \in \mathbb{Z}$: $a^m \cdot a^n = a^{m+n}$ y $\dfrac{a^m}{a^n} = a^{m-n}$.
\end{definicion}

\begin{formula}
\[ (a^m)^n = a^{m\cdot n} \qquad a^{-n} = \frac{1}{a^n} \]
\end{formula}

\begin{ejemplo}
¿Cuál es el valor de $\dfrac{2^{5}\cdot 4^{3}}{8^{3}}$?
\alternativa{A}{$2$}
\alternativa{B}{$8$}
\alternativa{C}{$4$}
\alternativa{D}{$16$}
\clave{C}
\begin{enumerate}
  \item $4^3 = 2^6$ y $8^3 = 2^9$.
  \item $\dfrac{2^{5}\cdot 2^{6}}{2^{9}} = 2^{11-9} = 2^2 = 4$.
\end{enumerate}
\end{ejemplo}

\begin{errorfrecuente}
Sumar los exponentes al elevar una potencia: $(2^2)^3 = 2^6$, no $2^5$.
\end{errorfrecuente}

\end{dato}
